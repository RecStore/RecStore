160 inline void RecordGpuCacheLookupOutcome(
161 int64_t num_keys, double hit_count, double request_count) {
162 if (num_keys < kGpuCacheBypassMinRows || request_count <= 0.0) {

166 if (hit_ratio < kGpuCacheLowHitRatio) {

177 }

211 inline void MaintainGpuCacheAfterUpdateNoThrow(
212 const torch::Tensor& keys,
213 const torch::Tensor& grads,
214 int64_t embedding_dim) {
215 (void)grads;
216 if (!gpu::IsGpuCacheEnabled()) {
217 return;
218 }
219 // When the adaptive bypass system is explicitly disabled, an external
220 // controller (e.g. BagPipe) owns cache invalidation: it pushes deltas the
221 // cache has already applied, so invalidating here (or clearing the whole
222 // cache when keys are on CPU) would needlessly evict valid entries.
223 if (!g_gpu_cache_lookup_bypass_enabled) {
224 return;
225 }
226 if (ShouldBypassGpuCacheMaintenance(keys.numel())) {
227 gpu::ResetLastGpuCacheProfile();
228 return;
229 }
230 if (gpu::CanUseGpuCache(keys, embedding_dim)) {
231 try {
232 gpu::InvalidateGpuCache(keys);
233 return;
234 } catch (const std::exception& e) {
235 LOG(WARNING) << "GPU cache invalidation failed after backend update "
236 "succeeded; clearing cache and continuing: "
237 << e.what();
238 } catch (...) {
239 LOG(WARNING) << "GPU cache invalidation failed after backend update "
240 "succeeded; clearing cache and continuing: "
241 << "unknown exception";
242 }
243 }
244 SafeClearGpuCacheNoThrow();
245 gpu::ResetLastGpuCacheProfile();
246 }
